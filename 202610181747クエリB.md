```dataviewjs
const toArr = v => v == null ? [] : (Array.isArray(v) ? v : (v.values ?? [v]));
const base = s => String(s).split("/").pop().replace(/\.md$/, "");

const pages = dv.pages('"items"')
  .where(p => !p.type || p.type === "preference")
  .sort(p => p.category, 'asc')
  .values;

const lines = [];
for (const p of pages) {
  const hubs = [...new Set(toArr(p.file.outlinks).map(l => base(l.path)))].join("、");
  const tags = toArr(p.my_tags).join("、");
  const text = await dv.io.load(p.file.path);
  const m = text.match(/## 主観\r?\n([\s\S]*?)(\r?\n## |$)/);
  const subj = m ? m[1].trim().replace(/\r?\n/g, " ") : "";
  lines.push([p.category ?? "", p.item ?? p.file.name, p.rating ?? "", hubs, tags, subj].join(" | "));
}

dv.el("pre", "カテゴリ | アイテム | 評価 | 抽象メタタグ | my_tags | 主観\n" + lines.join("\n"));
```
